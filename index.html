How to run
  1. Install Scilab (free) from scilab.org and open SciNotes (type scinotes in the console).
  2. Paste one program, save the file as name.sce.
  3. F5 runs the file. Ctrl+E runs only the selected lines.
  4. Output appears in the Console. Graphs open in a new window. input() prompts are answered in the Console.
  5. For CSV and save/load programs, set your working folder first (File Browser or cd).

// --- Linear regression: y = a*x + b ---
x = [1 2 3 4 5 6 7 8];           // row vectors for reglin
y = [2.1 4.2 5.9 8.1 10.2 11.8 14.1 16.0];
[a, b, sig] = reglin(x, y);

// --- Principal component analysis ---
// rows = observations, columns = variables
data = [2.5 2.4 1.0;
        0.5 0.7 0.3;
        2.2 2.9 1.1;
        1.9 2.2 0.9;
        3.1 3.0 1.4;
        2.3 2.7 1.0;
        2.0 1.6 0.8;
        1.0 1.1 0.5];
[facpr, comprinc, lambda, tsquare] = princomp(data);
disp(facpr);                      // principal axes (directions)
disp(lambda);                     // variance along each component
disp(lambda / sum(lambda) * 100); // % of variance explained
// Interpretation: if the first value of the last line is large
// (e.g. above 90 %), one component captures almost all the information,
// so the 3 variables can be reduced to 1 dimension with little loss.
// comprinc = data expressed in the new component axes.

Program 7: Menu-driven program with user-defined functions

// Functions must be defined BEFORE the menu code that calls them.
function f = fact_num(n)
    f = 1;
    for i = 2:n
        f = f * i;
    end
endfunction

function p = check_prime(n)
    p = %t;
    if n < 2 then
        p = %f;
    else
        for i = 2:floor(sqrt(n))
            if modulo(n, i) == 0 then
                p = %f;
                break;
            end
        end
    end
endfunction

function r = check_evenodd(n)
    if modulo(n, 2) == 0 then
        r = "Even";
    else
        r = "Odd";
    end
endfunction

while %t
    mprintf("\n1.Factorial 2.Prime check 3.Even/Odd 4.Exit\n");
    ch = input("Enter choice: ");
    if ch == 4 then
        mprintf("Goodbye!\n");
        break;
    end
    if ch >= 1 & ch <= 3 then
        n = input("Enter a number: ");
    end
    select ch
    case 1 then
        mprintf("Factorial of %d = %d\n", n, fact_num(n));
    case 2 then
        if check_prime(n) then
            mprintf("%d is prime\n", n);
        else
            mprintf("%d is not prime\n", n);
        end
    case 3 then
        mprintf("%d is %s\n", n, check_evenodd(n));
    else
        mprintf("Invalid choice\n");
    end
end
